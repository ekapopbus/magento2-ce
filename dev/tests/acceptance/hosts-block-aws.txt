# /etc/hosts snippet to block common AWS endpoints for local development
#
# WARNING: This redirects selected AWS hostnames to localhost (127.0.0.1).
# It prevents PHP/AWS SDK/other clients from reaching AWS. Use only for local
# development and testing. Do NOT commit /etc/hosts changes to shared images.
#
# To apply (run as root on the host):
#   sudo cp /etc/hosts /etc/hosts.mftf-block-aws.bak
#   sudo tee -a /etc/hosts < dev/tests/acceptance/hosts-block-aws.txt
#
# To revert (restore backup):
#   sudo cp /etc/hosts.mftf-block-aws.bak /etc/hosts
